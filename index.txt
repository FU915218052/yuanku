    // 绑定卡片图片悬停轮播
    function bindCardImgHover(cardEl) {
      const wrap = cardEl.querySelector('.img-wrap');
      if (!wrap) return;
      const imgs = wrap.querySelectorAll('img');
      if (imgs.length <= 1) return;

      let timer = null;
      let currentIndex = 0;

      const switchTo = (idx) => {
        imgs.forEach(img => img.classList.remove('active'));
        imgs[idx]?.classList.add('active');
        currentIndex = idx;
      };

      wrap.addEventListener('mouseenter', () => {
        if (imgs.length === 2) {
          switchTo(1);
        } else if (imgs.length > 2) {
          timer = setInterval(() => {
            currentIndex = (currentIndex + 1) % imgs.length;
            switchTo(currentIndex);
          }, 800);
        }
      });

      wrap.addEventListener('mouseleave', () => {
        if (timer) {
          clearInterval(timer);
          timer = null;
        }
        switchTo(0);
      });
    }

      if (bvMatch) {
        const bvid = 'BV' + bvMatch[1];
        return `<div class="video-embed"><iframe src="https://player.bilibili.com/player.html?bvid=${bvid}&page=1&high_quality=1&danmaku=0&autoplay=0" scrolling="no" border="0" frameborder="no" framespacing="0" allowfullscreen="true" sandbox="allow-top-navigation allow-same-origin allow-scripts allow-popups"></iframe></div>`;
      }
      if (url.startsWith('http')) {
        return `<div class="video-embed"><iframe src="${url}" scrolling="no" border="0" frameborder="no" framespacing="0" allowfullscreen="true"></iframe></div>`;
      }
      return '';
    }

    function initImageResizer() {
      const editor = document.getElementById('editorArea');
      if (!editor) return;

      if (resizer && resizer.parentNode) {
        resizer.parentNode.removeChild(resizer);
      }
      resizer = null;
      activeImg = null;

      resizer = document.createElement('div');
      resizer.className = 'img-resizer';
      const handles = ['tl','t','tr','l','r','bl','b','br'];
      handles.forEach(h => {
        const handle = document.createElement('div');
        handle.className = `resize-handle handle-${h}`;
        resizer.appendChild(handle);
      });
      editor.appendChild(resizer);

      editor.querySelectorAll('img').forEach(img => {
        img.onclick = (e) => {
          e.stopPropagation();
          // 自动选中当前图片，让超链接按钮能识别到
          const selection = window.getSelection();
          const range = document.createRange();
          range.selectNode(img);
          selection.removeAllRanges();
          selection.addRange(range);
          
          if (activeImg === img) {
            showResizer(img);
          } else {
            clearActiveImg();
            activeImg = img;
            img.classList.add('editor-img-active');
          }
        };
      });

      editor.onclick = (e) => {
        if (!e.target.closest('img')) {
          clearActiveImg();
          hideResizer();
        }
      };
    }

    function showResizer(img) {
      const rect = img.getBoundingClientRect();
      const editorRect = document.getElementById('editorArea').getBoundingClientRect();
      resizer.style.display = 'block';
      resizer.style.left = rect.left - editorRect.left + 'px';
      resizer.style.top = rect.top - editorRect.top + 'px';
      resizer.style.width = img.offsetWidth + 'px';
      resizer.style.height = img.offsetHeight + 'px';
      startRatio = img.naturalWidth / img.naturalHeight;

      const handles = resizer.querySelectorAll('.resize-handle');
      handles.forEach(handle => {
        handle.onmousedown = (e) => {
          e.stopPropagation();
          startX = e.clientX;
          startY = e.clientY;
          startWidth = img.offsetWidth;
          startHeight = img.offsetHeight;

          document.onmousemove = (e) => {
            let dx = e.clientX - startX;
            let dy = e.clientY - startY;
            if (handle.classList.contains('handle-l') || handle.classList.contains('handle-r')) dx *= 2;
            if (handle.classList.contains('handle-t') || handle.classList.contains('handle-b')) dy *= 2;
            
            let w = startWidth + dx;
            let h = w / startRatio;
            if (w < 50) w = 50;
            if (h < 50) h = 50;
            
            img.style.width = w + 'px';
            img.style.height = 'auto';
            resizer.style.width = w + 'px';
            resizer.style.height = img.offsetHeight + 'px';
          };

          document.onmouseup = () => {
            document.onmousemove = null;
            document.onmouseup = null;
          };
        };
      });
    }

    function hideResizer() {
      if (resizer) resizer.style.display = 'none';
    }

    function clearActiveImg() {
      if (activeImg) {
        activeImg.classList.remove('editor-img-active');
        activeImg = null;
      }
    }

          const realSrc = `images/${fileName}`;

          `;
          previewList.appendChild(item);

          item.querySelector('.img-preview-del').onclick = () => {
            const idx = currentImgs.indexOf(realSrc);
            if (idx > -1) currentImgs.splice(idx, 1);
            item.remove();
            imgValInput.value = JSON.stringify(currentImgs);
          };
        });

        imgValInput.value = JSON.stringify(currentImgs);
        upload.value = '';
      };

    function getPinyinFirstLetter(str) {
      const pinyinMap = {'一':'Y','二':'E','三':'S','四':'S','五':'W','六':'L','七':'Q','八':'B','九':'J','十':'S'};
      const hanReg = /[\u4e00-\u9fa5]/g;
      let result = "";
      for(let char of str){
        if(/[a-zA-Z]/g.test(char)) result += char.toUpperCase();
        else if(hanReg.test(char)) result += pinyinMap[char] || char;
      }
      return result;
    }

    function renderSidePanel() {
      const panel = document.getElementById('sidePanel');
      if (!panel) return;
      panel.innerHTML = '';
      
      sideItems.forEach((item, idx) => {
        const div = document.createElement('div');
        div.className = 'side-item';
        div.dataset.index = idx;
        div.innerText = item.name;
        // 应用自定义样式
        const style = item.style || {};
        if (style.color) div.style.color = style.color;
        if (style.bgColor) div.style.backgroundColor = style.bgColor;
        if (style.bold) div.style.fontWeight = 'bold';
        if (style.fontSize) div.style.fontSize = style.fontSize + 'px';
        div.innerHTML += `
          <div class="side-ctrl">
            <button class="side-edit-btn">编辑</button>
            <button class="side-del-btn">删除</button>
          </div>
        `;
        panel.appendChild(div);

        div.addEventListener('mouseenter', () => {
          if (isSidePopupFixed) return;
          showSidePopup(idx);
        });
        div.addEventListener('mouseleave', () => {
          if (isSidePopupFixed) return;
          hideSidePopup();
        });

        div.addEventListener('click', (e) => {
          if (e.target.closest('.side-ctrl')) return;
          e.stopPropagation();
          e.preventDefault();

          const popup = document.getElementById('sidePopup');

          // 已经固定 + 点的是当前同一条目：关闭弹窗
          if (isSidePopupFixed && currentSideItem === idx) {
            closeSidePopup();
            return;
          }

          // 其他情况：直接显示对应条目内容并保持固定
          isSidePopupFixed = true;
          showSidePopup(idx);
          popup.classList.add('show', 'side-popup-fixed');
          document.querySelectorAll('.side-item').forEach(i => i.classList.remove('active'));
          div.classList.add('active');
        });

        div.querySelector('.side-edit-btn').addEventListener('click', (e) => {
          e.stopPropagation();
          editSideItem(idx);
        });

        div.querySelector('.side-del-btn').addEventListener('click', (e) => {
          e.stopPropagation();
          if (!confirm('确定删除该条目？')) return;
          sideItems.splice(idx, 1);
          saveSideItems();
          renderSidePanel();
        });
      });

      const addBtn = document.createElement('div');
      addBtn.className = 'side-add-btn';
      addBtn.innerText = '+ 新增条目';
      addBtn.addEventListener('click', () => {
        sideItems.push({ name: '新条目', content: '<p>请编辑内容...</p>' });
        saveSideItems();
        renderSidePanel();
      });
      panel.appendChild(addBtn);
    }

// 显示右侧卡片详情预览
function showCardPreview(card) {
  const panel = document.getElementById('cardPreviewPanel');
  const imgs = JSON.parse(card.dataset.imgs || '[]');
  const links = JSON.parse(card.dataset.links || '[]');

  const firstImg = card.querySelector('.img-wrap img:first-child');
  const previewWrap = document.getElementById('previewIconWrap');

  if (firstImg && firstImg.complete && firstImg.naturalWidth > 0) {
    // 直接画到画布，不发新请求，视觉100%一致
    const canvas = document.createElement('canvas');
    canvas.width = firstImg.naturalWidth;
    canvas.height = firstImg.naturalHeight;
    const ctx = canvas.getContext('2d');
    ctx.drawImage(firstImg, 0, 0);
    
    previewWrap.innerHTML = '';
    previewWrap.appendChild(canvas);
    canvas.style.width = '70px';
    canvas.style.height = '70px';
    canvas.style.objectFit = 'cover';
    canvas.style.borderRadius = '8px';
    canvas.style.display = 'block';
  } else {
    document.getElementById('previewIcon').src = imgs[0] || '';
  }

  document.getElementById('previewTitle').innerText = card.dataset.name;
  document.getElementById('previewDesc').innerText = card.querySelector('.desc').innerText;
  
  // 渲染链接
  const linkWrap = document.getElementById('previewLinkWrap');
  linkWrap.innerHTML = '';
  if (links.length > 0) {
    links.forEach(link => {
      if (!link || (!link.href && !link.text)) return;
      const item = document.createElement('div');
      item.className = 'card-preview-link-item';
      item.innerHTML = `
        <span class="card-preview-link" data-href="${link.href || ''}">${link.text || link.href || '链接'}</span>
        <span class="card-preview-copy">复制</span>
      `;
      linkWrap.appendChild(item);

      item.querySelector('.card-preview-link').onclick = () => {
        if (link.href) window.open(link.href, '_blank');
      };
      item.querySelector('.card-preview-copy').onclick = (e) => {
        e.stopPropagation();
        navigator.clipboard.writeText(link.href || '').then(() => {
          item.querySelector('.card-preview-copy').innerText = '已复制';
          setTimeout(() => item.querySelector('.card-preview-copy').innerText = '复制', 1500);
        });
      };
    });
  }

  document.getElementById('previewContent').innerHTML = card.dataset.html || defaultHtml;
  renderSubGroups(card, document.getElementById('previewContent'));
  applyWatermark(document.getElementById('previewContent'));
  
  panel.classList.add('show');
}

      // 清理旧的背景图
      const oldBg = box.querySelector('.jump-bg-img');
      if (oldBg) oldBg.remove();

      const tipType = document.body.dataset.directTipType || 'text';
      const tipImg = document.body.dataset.directTipImg || '';

      if (tipType === 'image' && tipImg) {
        box.classList.add('img-mode');
        const bgImg = document.createElement('img');
        bgImg.className = 'jump-bg-img';
        bgImg.src = tipImg;
        bgImg.alt = '';
        box.insertBefore(bgImg, box.firstChild);
        tipEl.innerText = config.tip || '';
      } else {
        tipEl.innerText = config.tip || '⚠️ 即将跳转非本站网页，请注意账号密码财产安全';
      }

      nameEl.innerText = config.name || '';
      urlEl.innerText = config.url || '';
      jumpTargetUrl = config.url || '';
      let leftTime = config.time || 10;
      numEl.innerText = leftTime;

      modal.style.display = 'flex';

      if (jumpTimer) clearInterval(jumpTimer);
      jumpTimer = setInterval(() => {
        leftTime--;
        numEl.innerText = leftTime;
        if (leftTime <= 0) {
          clearInterval(jumpTimer);
          jumpTimer = null;
          window.open(jumpTargetUrl, '_blank');
          closeJumpModal();
        }
      }, 1000);
    }

    function bindSubDrag(sub) {
      sub.addEventListener('dragstart', (e) => {
        if (!isAdmin) {
          e.preventDefault();
          return;
        }
        draggingSub = sub;
        sub.style.opacity = '0.5';
        e.dataTransfer.effectAllowed = 'move';
      });
      sub.addEventListener('dragend', () => {
        draggingSub = null;
        document.querySelectorAll('.sub-card').forEach(s => s.style.opacity = '1');
      });
      sub.addEventListener('dragover', (e) => {
        if (!draggingSub || draggingSub === sub) {
          e.preventDefault();
          return;
        }
        e.preventDefault();
      });
      sub.addEventListener('drop', (e) => {
        e.preventDefault();
        if (!draggingSub || draggingSub === sub) return;

        const fromG = parseInt(draggingSub.dataset.gidx);
        const fromS = parseInt(draggingSub.dataset.sidx);
        const toG = parseInt(sub.dataset.gidx);
        const toS = parseInt(sub.dataset.sidx);
        const card = selectedCard;
        let groups = JSON.parse(card.dataset.subgroups);

        const moved = groups[fromG].cards.splice(fromS, 1)[0];
        if (fromG === toG) {
          groups[toG].cards.splice(toS, 0, moved);
        } else {
          groups[toG].cards.splice(toS, 0, moved);
          if (groups[fromG].cards.length === 0) groups.splice(fromG, 1);
        }

        card.dataset.subgroups = JSON.stringify(groups);
        refreshDetail(card);
      });
    }

      const up = groupEl.querySelector('.sub-group-up');
      const down = groupEl.querySelector('.sub-group-down');
      const idx = parseInt(groupEl.dataset.index);

      up.onclick = () => {
        let groups = JSON.parse(card.dataset.subgroups);
        if(idx <= 0) return;
        [groups[idx], groups[idx-1]] = [groups[idx-1], groups[idx]];
        card.dataset.subgroups = JSON.stringify(groups);
        refreshDetail(card);
      };
      down.onclick = () => {
        let groups = JSON.parse(card.dataset.subgroups);
        if(idx >= groups.length - 1) return;
        [groups[idx], groups[idx+1]] = [groups[idx+1], groups[idx]];
        card.dataset.subgroups = JSON.stringify(groups);
        refreshDetail(card);
      };
    }

    // 保存子卡片组的样式和数量（只读取详情弹窗里的设置，忽略预览窗口）
    function saveSubGroupsData() {
      if (!selectedCard) return;
      let groups = JSON.parse(selectedCard.dataset.subgroups || '[]');
      const detailBox = document.getElementById('detailContent');
      if (!detailBox) return;
      
      detailBox.querySelectorAll('.sub-card-group').forEach((groupEl, idx) => {
        if (!groups[idx]) return;
        const grid = groupEl.querySelector('.sub-card-grid');
        const slider = groupEl.querySelector('.sub-card-count-slider');
        const titleEl = groupEl.querySelector('.sub-group-title');
        const subtitleEl = groupEl.querySelector('.sub-group-subtitle');
        
        if (grid) groups[idx].style = grid.dataset.style || 'square';
        if (slider) groups[idx].count = parseInt(slider.value);

        canvas.width = firstImg.naturalWidth;
        canvas.height = firstImg.naturalHeight;
        const ctx = canvas.getContext('2d');
        ctx.drawImage(firstImg, 0, 0);

    // 自定义双击阈值：220ms，仅真实鼠标点击生效，减少快速单击误触双击
    let lastClickTime = 0;
    let lastClickX = 0;
    let lastClickY = 0;
    document.addEventListener('click', (e) => {
      // 仅处理真实鼠标点击，空格等脚本触发的点击不参与双击判定
      if (!e.isTrusted) return;
      
      const now = Date.now();
      const dx = Math.abs(e.clientX - lastClickX);
      const dy = Math.abs(e.clientY - lastClickY);
      
      // 220ms内、位置偏差小于5像素，判定为双击，手动触发dblclick事件
      if (now - lastClickTime <= 220 && dx < 5 && dy < 5) {
        e.target.dispatchEvent(new MouseEvent('dblclick', {
          bubbles: true,
          cancelable: true,
          clientX: e.clientX,
          clientY: e.clientY
        }));
      }
      
      lastClickTime = now;
      lastClickX = e.clientX;
      lastClickY = e.clientY;
    }, true);

      document.getElementById('fontSize')?.addEventListener('change', () => {

      document.getElementById('btnLink')?.addEventListener('click', function() {
        const linkUrl = prompt("请输入跳转链接", "https://");
        if (!linkUrl || linkUrl.trim() === "") return;
        const targetUrl = linkUrl.trim();

        const selection = window.getSelection();
        if (selection.rangeCount < 1) return;

        // 优先用你单击选中的图片
        let targetImg = activeImg;

        if (targetImg) {
          // 给图片包裹跳转链接，新窗口打开
          const linkWrap = document.createElement("a");
          linkWrap.href = targetUrl;
          linkWrap.target = "_blank";
          linkWrap.style.display = "inline-block";
          linkWrap.style.lineHeight = "0";
          linkWrap.style.textDecoration = "none";
          
          targetImg.parentNode.insertBefore(linkWrap, targetImg);
          linkWrap.appendChild(targetImg);
          
          // 重新选中图片，保持可调整大小状态
          const newRange = document.createRange();
          newRange.selectNode(linkWrap);
          selection.removeAllRanges();
          selection.addRange(newRange);
          initImageResizer();
        } else {
          // 选中的是文字，走原来的文字加链接逻辑
          document.execCommand("createLink", false, targetUrl);
          document.querySelectorAll("#editorArea a").forEach(a => {
            a.target = "_blank";
            a.style.textDecoration = "underline";
          });
        }
      });

      // 插入视频按钮
      document.getElementById('btnVideo')?.addEventListener('click', () => {
        const editor = document.getElementById('editorArea');
        if (!editor) return;
        const sel = window.getSelection();
        if (sel.rangeCount > 0) {
          editor._savedRange = sel.getRangeAt(0).cloneRange();
        }
        document.getElementById('videoUrlInput').value = '';
        document.getElementById('videoModal').style.display = 'flex';
      });

      document.getElementById('btnColor')?.addEventListener('input', (e) => {

      });

      // 背景色
      document.getElementById('btnBgColor')?.addEventListener('input', (e) => {
        document.execCommand('backColor', false, e.target.value);
      });
    }

            range.setStartAfter(img);

          } else {
            editor.appendChild(img);
          }
        });
        
        if (range) {
          sel.removeAllRanges();
          sel.addRange(range);
        }
        initImageResizer();
        imgInput.value = '';
      };
    };
